param([Parameter(Mandatory=$true)][string]$Suite,[Parameter(Mandatory=$true)][string]$Phase,[switch]$Script)
$ErrorActionPreference='Stop'
$s3Root=(Resolve-Path -LiteralPath (Join-Path $PSScriptRoot '../../..')).Path
$s3Engine=Join-Path $env:USERPROFILE 'Desktop/Godot_v4.7.1-stable_win64.exe/Godot_v4.7.1-stable_win64.exe'
. (Join-Path $s3Root 'tools/capture/city_process_guard.ps1')
$s3Active=@(Get-CimInstance Win32_Process | Where-Object {Test-CityCaptureGameProcess -Name $_.Name -CommandLine $_.CommandLine})
if ($s3Active.Count -gt 0) {throw 'Active Godot game/import exists; left untouched.'}
$s3Prefix=Join-Path $PSScriptRoot $Phase
if (Test-Path -LiteralPath ($s3Prefix+'-result.json')) {throw 'Use a fresh phase name to preserve evidence.'}
$s3OldApp=$env:APPDATA
$s3OldLocal=$env:LOCALAPPDATA
try {
  $s3Profile=Join-Path $PSScriptRoot ('profiles/'+$Phase)
  New-Item -ItemType Directory -Path $s3Profile -Force | Out-Null
  $env:APPDATA=$s3Profile
  $env:LOCALAPPDATA=$s3Profile
  $s3Args=@('--headless','--path',('"'+$s3Root+'"'),'--log-file',('"'+$s3Prefix+'-engine.log"'))
  if($Script) {$s3Args+=@('--script',('res://tools/tests/'+$Suite+'.gd'))} else {$s3Args+=('res://tools/tests/'+$Suite+'.tscn')}
  $s3Args+=@('--quit-after','3000')
  $s3Process=Start-Process -FilePath $s3Engine -ArgumentList $s3Args -WindowStyle Hidden -PassThru -RedirectStandardOutput ($s3Prefix+'-stdout.log') -RedirectStandardError ($s3Prefix+'-stderr.log')
} finally {$env:APPDATA=$s3OldApp;$env:LOCALAPPDATA=$s3OldLocal}
$s3Timeout=-not $s3Process.WaitForExit(300000)
if ($s3Timeout) {Stop-Process -Id $s3Process.Id -Force;$s3Process.WaitForExit()}
$s3Errors=@(Get-Content -LiteralPath ($s3Prefix+'-stderr.log') | Where-Object {$_ -match 'ERROR:|WARNING:|SCRIPT ERROR|Parse Error'})
[ordered]@{suite=$Suite;exit=$s3Process.ExitCode;timeout=$s3Timeout;diagnostics=$s3Errors} | ConvertTo-Json -Depth 4 | Set-Content -LiteralPath ($s3Prefix+'-result.json') -Encoding utf8
Get-Content -LiteralPath ($s3Prefix+'-stdout.log') | Select-String 'Test:|passes=|failures=' | Select-Object -Last 2 | ForEach-Object {$_.Line}
Get-Content -LiteralPath ($s3Prefix+'-stderr.log') -Tail 5
Get-Content -LiteralPath ($s3Prefix+'-result.json')
if ($s3Timeout -or $s3Process.ExitCode -ne 0 -or $s3Errors.Count -gt 0) {exit 1}
exit 0
